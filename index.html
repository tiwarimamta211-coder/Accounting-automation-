Monthly Income/Expense Summary 4. Invoice Generator 5. Final Report in Excel ============================================================ """ import os import glob import pandas as pd from openpyxl import Workbook, load_workbook from openpyxl.styles import Font, PatternFill, Alignment, Border, Side from openpyxl.utils import get_column_letter from datetime import datetime # ───────────────────────────────────────────── # CONFIG — Apni details yahan bharen # ───────────────────────────────────────────── COMPANY_NAME = "Aapki Company Ka Naam" COMPANY_ADDRESS = "Address, Shehar, State - PIN" COMPANY_GSTIN = "GSTIN123456789" INPUT_FOLDER = "input_files" # Yahan apni Excel files rakhein OUTPUT_FILE = "accounting_report.xlsx" INVOICE_PREFIX = "INV" GST_RATE = 18 # GST % (change as needed: 5, 12, 18, 28) # ───────────────────────────────────────────── # STYLES # ───────────────────────────────────────────── HEADER_FILL = PatternFill("solid", start_color="1F4E79") SUB_FILL = PatternFill("solid", start_color="2E75B6") ALT_FILL = PatternFill("solid", start_color="D6E4F0") TOTAL_FILL = PatternFill("solid", start_color="FFF2CC") WHITE_FILL = PatternFill("solid", start_color="FFFFFF") HEADER_FONT = Font(name="Arial", bold=True, color="FFFFFF", size=11) SUBHEAD_FONT = Font(name="Arial", bold=True, color="FFFFFF", size=10) TOTAL_FONT = Font(name="Arial", bold=True, color="000000", size=10) NORMAL_FONT = Font(name="Arial", size=10) CENTER = Alignment(horizontal="center", vertical="center") LEFT = Alignment(horizontal="left", vertical="center") RIGHT = Alignment(horizontal="right", vertical="center") thin = Side(style="thin", color="BFBFBF") BORDER = Border(left=thin, right=thin, top=thin, bottom=thin) def style_cell(cell, font=None, fill=None, align=None, border=True, num_format=None): if font: cell.font = font if fill: cell.fill = fill if align: cell.alignment = align if border: cell.border = BORDER if num_format: cell.number_format = num_format def header_row(ws, row, cols, texts, fill=None, font=None): fill = fill or HEADER_FILL font = font or HEADER_FONT for col, text in zip(cols, texts): c = ws.cell(row=row, column=col, value=text) style_cell(c, font=font, fill=fill, align=CENTER) # ───────────────────────────────────────────── # 1. MERGE MULTIPLE EXCEL FILES # ───────────────────────────────────────────── def merge_excel_files(folder=INPUT_FOLDER): files = glob.glob(os.path.join(folder, "*.xlsx")) + glob.glob(os.path.join(folder, "*.csv")) if not files: print(f"[!] '{folder}' folder mein koi file nahi mili. Sample data use ho raha hai.") return generate_sample_data() dfs = [] for f in files: try: df = pd.read_excel(f) if f.endswith(".xlsx") else pd.read_csv(f) df["Source_File"] = os.path.basename(f) dfs.append(df) print(f" ✓ {os.path.basename(f)} — {len(df)} rows") except Exception as e: print(f" ✗ {os.path.basename(f)} skip: {e}") if not dfs: return generate_sample_data() merged = pd.concat(dfs, ignore_index=True) # Standardise common column names rename_map = {} for col in merged.columns: lower = col.strip().lower() if lower in ["amount", "amt", "value", "total"]: rename_map[col] = "Amount" elif lower in ["date", "invoice date", "bill date"]: rename_map[col] = "Date" elif lower in ["description", "desc", "item", "particulars"]: rename_map[col] = "Description" elif lower in ["type", "category", "head"]: rename_map[col] = "Type" elif lower in ["party", "customer", "vendor", "client"]: rename_map[col] = "Party" merged.rename(columns=rename_map, inplace=True) return merged def df # ───────────────────────────────────────────── # 2. GST CALCULATION # ───────────────────────────────────────────── def calculate_gst(df, rate=GST_RATE): gst = df.copy() if "Amount" not in gst.columns: print("[!] 'Amount' column nahi mila.") return gst half = rate / 2 gst["Base_Amount"] = gst["Amount"] gst["CGST_%"] = half gst["SGST_%"] = half gst["IGST_%"] = 0.0 gst["CGST_Amt"] = (gst["Amount"] * half / 100).round(2) gst["SGST_Amt"] = (gst["Amount"] * half / 100).round(2) gst["IGST_Amt"] = 0.0 gst["Total_GST"] = gst["CGST_Amt"] + gst["SGST_Amt"] gst["Grand_Total"] = gst["Amount"] + gst["Total_GST"] return gst # ───────────────────────────────────────────── # 3. MONTHLY SUMMARY # ───────────────────────────────────────────── def monthly_summary(df): df = df.copy() if "Date" in df.columns: df["Date"] = pd.to_datetime(df["Date"], errors="coerce") df["Month"] = df["Date"].dt.strftime("%Y-%m") else: df["Month"] = "Unknown" if "Type" not in df.columns: df["Type"] = "Income" df["Amount"] = pd.to_numeric(df["Amount"], errors="coerce").fillna(0) income = df[df["Type"].str.lower() == "income"].groupby("Month")["Amount"].sum() expense = df[df["Type"].str.lower() == "expense"].groupby("Month")["Amount"].sum() summary = pd.DataFrame({"Income": income, "Expense": expense}).fillna(0) summary["Net_Profit"] = summary["Income"] - summary["Expense"] summary["Profit_Margin"] = (summary["Net_Profit"] / summary["Income"].replace(0, 1) * 100).round(2) return summary.reset_index() # ───────────────────────────────────────────── # 4. INVOICE GENERATOR # ───────────────────────────────────────────── SAMPLE_INVOICES = [ {"Party": "Client A", "Items": [{"desc": "Web Development", "qty": 1, "rate": 50000}, {"desc": "SEO Services", "qty": 2, "rate": 5000}]}, {"Party": "Client B", "Items": [{"desc": "Mobile App Design","qty": 1, "rate": 80000}]}, {"Party": "Client C", "Items": [{"desc": "Consulting", "qty": 5, "rate": 3000}, {"desc": "Report Writing", "qty": 2, "rate": 2500}]}, ] def write_invoice_sheet(wb, inv_num, party, items, gst_rate, date_str): sheet_name = f"Invoice-{inv_num}" ws = wb.create_sheet(sheet_name) # Company header ws.merge_cells("A1:F1") ws["A1"] = COMPANY_NAME style_cell(ws["A1"], Font(name="Arial", bold=True, size=16, color="1F4E79"), align=CENTER, border=False) ws.merge_cells("A2:F2") ws["A2"] = COMPANY_ADDRESS style_cell(ws["A2"], NORMAL_FONT, align=CENTER, border=False) ws.merge_cells("A3:F3") ws["A3"] = f"GSTIN: {COMPANY_GSTIN}" style_cell(ws["A3"], NORMAL_FONT, align=CENTER, border=False) # Invoice details ws["A5"] = "Invoice No:" ws["B5"] = f"{INVOICE_PREFIX}-{inv_num:04d}" ws["D5"] = "Date:" ws["E5"] = date_str ws["A6"] = "Bill To:" ws["B6"] = party ws["D6"] = "GST Rate:" ws["E6"] = f"{gst_rate}%" for cell in ["A5","B5","D5","E5","A6","B6","D6","E6"]: style_cell(ws[cell], NORMAL_FONT, border=False) ws["A5"].font = ws["A6"].font = ws["D5"].font = ws["D6"].font = Font(name="Arial", bold=True, size=10) # Table header row = 8 cols = ["#", "Description", "Qty", "Rate (₹)", "Amount (₹)", "GST (₹)"] header_row(ws, row, range(1, 7), cols) # Items subtotal = 0 for i, item in enumerate(items, 1): row += 1 amt = item["qty"] * item["rate"] subtotal += amt fill = ALT_FILL if i % 2 == 0 else WHITE_FILL for col, val in zip(range(1,7), [i, item["desc"], item["qty"], item["rate"], amt, round(amt * gst_rate / 100, 2)]): c = ws.cell(row=row, column=col, value=val) style_cell(c, NORMAL_FONT, fill, align=RIGHT if col > 2 else LEFT) # Totals row += 1 gst_amt = round(subtotal * gst_rate / 100, 2) grand_total = subtotal + gst_amt for label, value, col_start in [ ("Subtotal", subtotal, 4), ("GST Amount", gst_amt, 4), ("Grand Total", grand_total, 4), ]: row += 1 c_label = ws.cell(row=row, column=col_start, value=label) c_value = ws.cell(row=row, column=col_start + 1, value=value) style_cell(c_label, TOTAL_FONT, TOTAL_FILL, align=RIGHT) style_cell(c_value, TOTAL_FONT, TOTAL_FILL, align=RIGHT, num_format='₹#,##0.00') # Column widths ws.column_dimensions["A"].width = 5 ws.column_dimensions["B"].width = 35 ws.column_dimensions["C"].width = 8 ws.column_dimensions["D"].width = 15 ws.column_dimensions["E"].width = 15 ws.column_dimensions["F"].width = 15 return grand_total # ───────────────────────────────────────────── # 5. BUILD FINAL REPORT # ───────────────────────────────────────────── def build_report(df, gst_df, summary, wb, invoices_total): today = datetime.today().strftime("%d %b %Y") # ── Sheet 1: Merged Data ────────────────── ws1 = wb.active ws1.title = "All Transactions" ws1.merge_cells("A1:H1") ws1["A1"] = f"ALL TRANSACTIONS — {COMPANY_NAME}" style_cell(ws1["A1"], HEADER_FONT, HEADER_FILL, CENTER, border=False) cols = list(df.columns)[:8] header_row(ws1, 2, range(1, len(cols)+1), cols) for r_idx, row in enumerate(df.itertuples(index=False), 3): fill = ALT_FILL if r_idx % 2 == 0 else WHITE_FILL for c_idx, val in enumerate(row[:8], 1): cell = ws1.cell(row=r_idx, column=c_idx, value=val) style_cell(cell, NORMAL_FONT, fill, LEFT) for i, col in enumerate(cols, 1): ws1.column_dimensions[get_column_letter(i)].width = 20 # ── Sheet 2: GST Sheet ──────────────────── ws2 = wb.create_sheet("GST Calculation") ws2.merge_cells("A1:I1") ws2["A1"] = f"GST CALCULATION SHEET — {GST_RATE}% Rate" style_cell(ws2["A1"], HEADER_FONT, HEADER_FILL, CENTER, border=False) gst_cols = ["Date","Description","Party","Type","Base_Amount","CGST_Amt","SGST_Amt","IGST_Amt","Grand_Total"] available = [c for c in gst_cols if c in gst_df.columns] header_row(ws2, 2, range(1, len(available)+1), available) for r_idx, row in enumerate(gst_df[available].itertuples(index=False), 3): fill = ALT_FILL if r_idx % 2 == 0 else WHITE_FILL for c_idx, val in enumerate(row, 1): cell = ws2.cell(row=r_idx, column=c_idx, value=val) is_money = available[c_idx-1] in ["Base_Amount","CGST_Amt","SGST_Amt","IGST_Amt","Grand_Total"] style_cell(cell, NORMAL_FONT, fill, RIGHT if is_money else LEFT, num_format='#,##0.00' if is_money else None) last_data = 2 + len(gst_df) total_row = last_data + 2 ws2.cell(row=total_row, column=len(available)-1, value="TOTAL:").font = TOTAL_FONT ws2.cell(row=total_row, column=len(available)-1).fill = TOTAL_FILL total_cell = ws2.cell(row=total_row, column=len(available), value=f"=SUM({get_column_letter(len(available))}3:{get_column_letter(len(available))}{last_data})") style_cell(total_cell, TOTAL_FONT, TOTAL_FILL, RIGHT, num_format='#,##0.00') for i in range(1, len(available)+1): ws2.column_dimensions[get_column_letter(i)].width = 18 # ── Sheet 3: Monthly Summary ────────────── ws3 = wb.create_sheet("Monthly Summary") ws3.merge_cells("A1:E1") ws3["A1"] = "MONTHLY INCOME & EXPENSE SUMMARY" style_cell(ws3["A1"], HEADER_FONT, HEADER_FILL, CENTER, border=False) s_cols = ["Month", "Income", "Expense", "Net_Profit", "Profit_Margin"] header_row(ws3, 2, range(1, 6), s_cols) for r_idx, row in enumerate(summary.itertuples(index=False), 3): fill = ALT_FILL if r_idx % 2 == 0 else WHITE_FILL for c_idx, val in enumerate(row[:5], 1): cell = ws3.cell(row=r_idx, column=c_idx, value=val) is_pct = s_cols[c_idx-1] == "Profit_Margin" is_money= s_cols[c_idx-1] in ["Income","Expense","Net_Profit"] fmt = '0.00"%"' if is_pct else ('#,##0.00' if is_money else None) style_cell(cell, NORMAL_FONT, fill, RIGHT if c_idx > 1 else LEFT, num_format=fmt) # Totals row tr = 2 + len(summary) + 1 ws3.cell(row=tr, column=1, value="TOTAL").font = TOTAL_FONT ws3.cell(row=tr, column=1).fill = TOTAL_FILL for col in range(2, 5): c = ws3.cell(row=tr, column=col, value=f"=SUM({get_column_letter(col)}3:{get_column_letter(col)}{2+len(summary)})") style_cell(c, TOTAL_FONT, TOTAL_FILL, RIGHT, num_format='#,##0.00') for i in range(1, 6): ws3.column_dimensions[get_column_letter(i)].width = 20 # ── Sheet 4: Dashboard ──────────────────── ws4 = wb.create_sheet("Dashboard") ws4.merge_cells("A1:D1") ws4["A1"] = f"ACCOUNTING DASHBOARD — {today}" style_cell(ws4["A1"], Font(name="Arial", bold=True, size=14, color="FFFFFF"), HEADER_FILL, CENTER, border=False) income_total = summary["Income"].sum() expense_total = summary["Expense"].sum() net_profit = income_total - expense_total kpis = [ ("Total Income", f"₹{income_total:,.2f}", "2E75B6"), ("Total Expense", f"₹{expense_total:,.2f}", "C00000"), ("Net Profit", f"₹{net_profit:,.2f}", "375623" if net_profit >= 0 else "C00000"), ("Total GST Payable", f"₹{gst_df['Total_GST'].sum():,.2f}" if 'Total_GST' in gst_df.columns else "N/A", "7030A0"), ("Total Invoiced", f"₹{invoices_total:,.2f}", "1F4E79"), ("Transactions", str(len(df)), "4A4A4A"), ] for i, (label, value, color) in enumerate(kpis, 1): row = i + 2 lc = ws4.cell(row=row, column=1, value=label) vc = ws4.cell(row=row, column=2, value=value) lc.font = Font(name="Arial", bold=True, size=11) vc.font = Font(name="Arial", bold=True, size=12, color=color) lc.fill = vc.fill = ALT_FILL if i % 2 == 0 else WHITE_FILL lc.border = vc.border = BORDER lc.alignment = LEFT vc.alignment = RIGHT ws4.column_dimensions["A"].width = 25 ws4.column_dimensions["B"].width = 25 # Make Dashboard first wb.move_sheet("Dashboard", offset=-wb.index(wb["Dashboard"])) # ───────────────────────────────────────────── # MAIN # ───────────────────────────────────────────── def main(): print("\n" + "="*55) print(" ACCOUNTING AUTOMATION TOOLKIT") print("="*55) #
